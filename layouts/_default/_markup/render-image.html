{{/* render-image hook - If image link is a page resource (relative path and present in page bundle), use Hugo image processing to create resized versions and output responsive srcset. - If image link is absolute (starts with /) or resource not found, fall back to outputting the original destination. - Always add `post-image` class so CSS from `static/css/custom.css` applies. */}} {{ $src := .Destination }} {{ $alt := .Text | safeHTMLAttr }} {{ $title := .Title | safeHTMLAttr }} {{ $class := "post-image" }} {{ if strings.HasPrefix $src "/" }} {{ else }} {{ $res := .Page.Resources.GetMatch (printf "**%s" $src) }} {{ if $res }} {{ $small := $res.Resize "600x" }} {{ $large := $res.Resize "1024x" }} {{ else }} {{ end }} {{ end }}
